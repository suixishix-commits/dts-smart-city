# 智慧城市数字孪生 · 实施计划 01：工程初始化、框架与城市总览

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 从模板建出新工程 `dts-smart-city`，搭好通用框架（顶栏、面板件、场景控制器、数据层），并交付第一个完整模块「城市总览」（人口密度峰值变化、热点事件、摄像头、时间轴、图例、工具栏）。

**Architecture:** 一个常驻 DTS 场景 + 路由切换模块。面板只调本模块的场景控制器；场景控制器通过 `SceneApi` 接口操作 DTS，并由基类 `ModuleScene` 登记、统一清理场景对象。数据全部来自 `api/` 层的 mock。

**Tech Stack:** Vue 3.5 + TypeScript + Vite 7 + Pinia + ECharts 6 + Element Plus（模板自带）；新增 Vitest + @vue/test-utils。DTS SDK 由 `http://127.0.0.1:8090/libac` 提供。

**相关文档**
- 设计文档：`docs/2026-11-13-智慧城市数字孪生系统-设计文档.md`（Task 1 迁入新工程后的路径）
- 页面设计稿：https://claude.ai/artifact/KsiifD2a9gwZxWf4qapcqC
- DTS 二次开发文档：http://127.0.0.1:8090/locale/zh/doc/index.html

**范围说明**
- 本计划只覆盖设计文档实施步骤的阶段 0–2。智慧交通、公共安全、生态环境、城市生命线各自另写计划（02–05），沿用本计划建立的约定。
- 本计划只用 DTS 原生接口（marker / polygon / heatmap / camera），不依赖特效 PAK。
- 摄像头画面本期是占位，不接真实视频流。
- 通用工具栏本期实现：视角复位、昼夜切换、天气模拟、全屏。图层管理、城市巡航、测量、截图本期显示为禁用，后续计划实现。

**已知事实（来自模板调研，执行时不必重新确认）**
- 模板 `package.json` 有 `test:unit: vitest` 脚本，但**没有安装 vitest**。
- 模板当前 `vue-tsc` 有 7 个错误，全部在自动生成的 `components.d.ts`；删除演示代码并重新生成后应为 0。
- `postcss-pxtorem` 会把 SFC 样式里的 px 转成 rem（1920 宽下 1rem = 10px），所以**样式一律写在 `<style>` 里按 1920 设计稿写 px，不写内联 style**。
- `vite.config.ts` 通过 `additionalData` 向每个 scss 注入 `@/styles/variables.scss`，变量可直接用。
- `unplugin-auto-import` 自动导入 `vue` / `vue-router` / `pinia` 的 API 和 `src/utils/**` 的导出；`unplugin-vue-components` 自动注册 `src/components`、`src/views` 下的组件。**单元测试文件里一律显式 import**。
- 场景就绪时 `EventDispatcher` 会 `emitter.emit('SceneOnReady', ...)`；点击 marker 时会 `emitter.emit(<marker 的 groupId>, new MittEvent(groupId, { id, userData, ... }))`，并对 `userData` 做 `JSON.parse`，所以 **marker 的 `userData` 必须是合法 JSON 字符串**。
- `__xapi.camera.set(x, y, z, pitch, yaw, flyTime)`；`__xapi.coord.gcs2pcs(coords, 1)` 把 WGS84 经纬度转成场景投影坐标，返回 `{ result: 0, coordinates }`。
- `__xapi.heatmap.addByHeatPoints({ id, bbox, range, data: [{ id, coordinate, radius, heatValue }] })`、`heatmap.delete(ids)`、`show/hide(ids)`。

Create `src/scene/ModuleScene.ts`：

```ts
import type { SceneApi, SceneObjectType } from './SceneApi'

interface Layer {
  type: SceneObjectType
  ids: string[]
  visible: boolean
}

/**
 * 模块场景控制器基类。
 * 子类在 onEnter 里通过 add() 创建对象；基类登记每个对象，leave() 时统一删除。
 * token 用来丢弃过期的进入流程：离开后（或再次进入后）旧流程的 add() 不再生效。
 */
export abstract class ModuleScene {
  private layers = new Map<string, Layer>()
  private generation = 0
  private active = false

  constructor(protected readonly api: SceneApi) {}

  /** 子类实现：设置相机、创建对象。所有 add() 调用都要传入收到的 token */
  protected abstract onEnter(token: number): Promise<void>
  /** 子类可覆盖：停止定时器、取消订阅 */
  protected async onLeave(): Promise<void> {}

  async enter(): Promise<void> {
    const token = ++this.generation
    this.active = true
    await this.onEnter(token)
  }

  async leave(): Promise<void> {
    this.active = false
    this.generation++
    await this.onLeave()
    const layers = [...this.layers.values()]
    this.layers.clear()
    for (const layer of layers) {
      if (layer.ids.length) await this.api.remove(layer.type, layer.ids)
    }
  }

  protected isCurrent(token: number): boolean {
    return this.active && token === this.generation
  }

  /** 添加一组对象并登记到名为 layer 的图层 */
  protected async add(token: number, type: SceneObjectType, objects: { id: string }[], layer: string): Promise<void> {
    if (!this.isCurrent(token) || objects.length === 0) return
    const entry = this.layers.get(layer) ?? { type, ids: [], visible: true }
    entry.ids.push(...objects.map(o => o.id))
    this.layers.set(layer, entry)
    await this.api.add(type, objects)
    if (!entry.visible) await this.api.setVisible(type, entry.ids, false)
  }

  /** 用新对象整体替换某个图层（用于随时间变化的图层） */
  async replaceLayer(layer: string, type: SceneObjectType, objects: { id: string }[]): Promise<void> {
    if (!this.active) return
    const old = this.layers.get(layer)
    const visible = old ? old.visible : true
    if (old?.ids.length) await this.api.remove(old.type, old.ids)
    this.layers.set(layer, { type, ids: objects.map(o => o.id), visible })
    if (objects.length) {
      await this.api.add(type, objects)
      if (!visible) await this.api.setVisible(type, objects.map(o => o.id), false)
    }
  }

  async setLayerVisible(layer: string, visible: boolean): Promise<void> {
    const entry = this.layers.get(layer)
    if (!entry) return
    entry.visible = visible
    if (entry.ids.length) await this.api.setVisible(entry.type, entry.ids, visible)
  }

  objectCount(): number {
    let n = 0
    this.layers.forEach(l => (n += l.ids.length))
    return n
  }
}
```

import type { SceneApi, SceneObjectType } from './SceneApi'

/** SceneApi 的 DTS 实现，直接调用 __xapi。只在场景就绪后使用。 */
export class DtsSceneApi implements SceneApi {
  async add(type: SceneObjectType, objects: { id: string }[]): Promise<void> {
    if (type === 'heatmap') {
      for (const o of objects) await __xapi.heatmap.addByHeatPoints(o)
      return
    }
    await __xapi[type].add(objects)
  }

  async remove(type: SceneObjectType, ids: string[]): Promise<void> {
    await __xapi[type].delete(ids)
  }

  async setVisible(type: SceneObjectType, ids: string[], visible: boolean): Promise<void> {
    if (visible) await __xapi[type].show(ids)
    else await __xapi[type].hide(ids)
  }

  async setCamera(view: number[], flyTime = 2): Promise<void> {
    await __xapi.camera.set(view[0], view[1], view[2], view[3], view[4], flyTime)
  }

  async focus(type: SceneObjectType, id: string, distance = 600): Promise<void> {
    await __xapi[type].focus(id, distance, 2)
  }

  async project(lonLat: number[][]): Promise<number[][]> {
    if (lonLat.length === 0) return []
    const res = await __xapi.coord.gcs2pcs(lonLat, 1)
    if (res.result !== 0) throw new Error('坐标转换失败：' + res.resultMessage)
    return res.coordinates
  }
}

/** 场景就绪后 resolve(true)；连接失败时 resolve(false)。 */
export function whenSceneReady(): Promise<boolean> {
  const state = store.dtsStore.sceneState
  if (state === 'ready') return Promise.resolve(true)
  if (state === 'failed') return Promise.resolve(false)
  return new Promise(resolve => {

    }

import type { District, HeatPoint, Hotspot } from '@/types/smartCity'

/** 主岛在岛人口（万人）一天内的控制点：[小时, 人口] */
const CURVE: [number, number][] = [
  [0, 62], [3, 58], [6, 61], [8, 74], [10, 79], [12, 80],
  [14.5, 78.6], [16, 82], [18, 88], [19.5, 91.2], [21, 84], [24, 66]
]
/** 密度基准时刻的人口（12:00） */
const BASELINE = 80

/** 某时刻的在岛人口（万人），控制点之间线性插值 */
export function islandPopulationAt(hour: number): number {
  const h = Math.min(24, Math.max(0, hour))
  for (let i = 1; i < CURVE.length; i++) {
    const [h0, v0] = CURVE[i - 1]
    const [h1, v1] = CURVE[i]
    if (h <= h1) return v0 + ((v1 - v0) * (h - h0)) / (h1 - h0)
  }
  return CURVE[CURVE.length - 1][1]
}

/** 一天中的峰值时刻与人口 */
export function populationPeak(): { hour: number; value: number } {
  const top = CURVE.reduce((a, b) => (b[1] > a[1] ? b : a))
  return { hour: top[0], value: top[1] }
}

/** 每半小时一个点的全天序列：[小时, 人口] */
export function populationSeries(): [number, number][] {
  const out: [number, number][] = []
  for (let i = 0; i <= 48; i++) {
    const h = i / 2
    out.push([h, Math.round(islandPopulationAt(h) * 10) / 10])
  }
  return out
}

/** 片区某时刻的人口密度（万人/km²） */
export function districtDensityAt(district: District, hour: number): number {
  return (district.baseDensity * islandPopulationAt(hour)) / BASELINE
}

/** 某时刻的热力点：每个片区中心一个点，每个热点事件一个点 */
export function heatPointsAt(hour: number, districts: District[], hotspots: Hotspot[]): HeatPoint[] {
  const points: HeatPoint[] = districts.map(d => ({
    id: 'heat-' + d.id,
    position: d.center,
    radius: 900,
    value: districtDensityAt(d, hour)
  }))
  hotspots.forEach(h => {
    points.push({ id: 'heat-' + h.id, position: h.position, radius: 450, value: (h.heat / 100) * 3 })
  })
  return points
}

import type { Alarm } from '@/types/smartCity'
import { defineStore } from 'pinia'

export const alarmStore = defineStore('alarmStore', {
  state: () => ({
    alarms: [] as Alarm[]
  }),
  getters: {
    /** 等级高的在前；同等级时间新的在前 */
    sorted: (state): Alarm[] => [...state.alarms].sort((a, b) => a.level - b.level || b.time.localeCompare(a.time)),
    total: (state): number => state.alarms.length
  },
  actions: {
    setAlarms(alarms: Alarm[]) {
      this.alarms = alarms
    },
    top(n: number): Alarm[] {
      return this.sorted.slice(0, n)
    }
  }
})

export default alarmStore

import { heatPointsAt } from '@/api/mock/population'
import { ModuleScene } from '@/scene/ModuleScene'
import type { SceneApi } from '@/scene/SceneApi'
import type { OverviewData } from '@/types/smartCity'

export interface OverviewSceneOptions {
  /** [x, y, z, pitch, yaw]；为 null 时自动定位到片区 */
  cameraView: number[] | null
  /** 热力图垂直范围 [minZ, maxZ] */
  heatmapZ: [number, number]
}

const HEATMAP_ID = 'population-heat'
/** 热力值范围，万人/km² */
const HEAT_RANGE = [0, 3]

// RGBA，取值 0–1
const WHITE = [1, 1, 1, 1]
const PANEL = [0.07, 0.09, 0.12, 0.92]
const BLUE = [0.3, 0.6, 1, 0.9]
const HOT = [1, 0.42, 0.42, 0.95] // 热度 ≥ 90
const WARM = [1, 0.66, 0.3, 0.95] // 热度 ≥ 75
const MILD = [1, 0.83, 0.23, 0.95]

function heatColor(heat: number): number[] {
  return heat >= 90 ? HOT : heat >= 75 ? WARM : MILD
}

/** 文字标注的公共样式（不带图片）。字段含义见 DTS 文档 Marker.add */
function textMarker(id: string, groupId: string, coordinate: number[], text: string, background: number[], userId: string) {
  return {
    id,
    groupId,
    userData: JSON.stringify({ id: userId }),
    coordinate,
    coordinateType: 0,
    range: [1, 1000000],
    imagePath: '',
    imageSize: [0, 0],
    fixedSize: true,
    text,
    useTextAnimation: false,
    textRange: [1, 1000000],
    textOffset: [0, 0],
    textBackgroundColor: background,
    fontSize: 14,
    fontColor: WHITE,
    showLine: true,
    lineSize: [2, 30],
    lineColor: background,
    lineOffset: [0, 0],
    autoHeight: true,
    displayMode: 2,
    occlusionCull: false
  }
}

export class OverviewScene extends ModuleScene {
  private data: OverviewData | null = null
  private hour = 12
  /** 经纬度字符串 → 投影坐标，避免重复转换 */
  private projected = new Map<string, number[]>()

  constructor(
    api: SceneApi,
    private readonly options: OverviewSceneOptions
  ) {
    super(api)
  }

  /** 进入前设置数据和初始小时 */
  setData(data: OverviewData, hour: number) {
    this.data = data
    this.hour = hour
  }

  protected async onEnter(token: number): Promise<void> {
    const data = this.data
    if (!data) return
    await this.projectAll(data)

    await this.add(
      token,
      'polygon',
      data.districts.map(d => ({
        id: 'district-' + d.id,
        groupId: 'sc_district',
        coordinates: d.boundary.map(p => this.at(p)),
        coordinateType: 0,
        range: [1, 1000000],
        color: [0.3, 0.6, 1, 0.06],
        frameColor: [0.79, 0.82, 0.86, 0.8],
        frameThickness: 2,
        intensity: 1,
        style: 0,
        depthTest: false
      })),
      'district'
    )

    await this.add(token, 'heatmap', [this.buildHeatmap(data)], 'heat')

    await this.add(
      token,
      'marker',
      data.hotspots.map((h, i) => textMarker('hotspot-' + h.id, 'sc_hotspot', this.at(h.position), `${i + 1}  ${h.shortName}`, heatColor(h.heat), h.id)),
      'hotspot'
    )

    await this.add(
      token,
      'marker',
      data.cameras.map(c => textMarker('camera-' + c.id, 'sc_camera', this.at(c.position), c.id, c.online ? BLUE : PANEL, c.id)),
      'camera'
    )

    if (!this.isCurrent(token)) return
    if (this.options.cameraView) {
      await this.api.setCamera(this.options.cameraView)
    } else if (data.districts.length) {
      await this.api.focus('polygon', 'district-' + data.districts[0].id, 6000)
    }
  }

  /** 切换到某个小时：整体替换热力图 */
  async setHour(hour: number): Promise<void> {
    this.hour = hour
    if (!this.data) return
    await this.replaceLayer('heat', 'heatmap', [this.buildHeatmap(this.data)])
  }

  async focusHotspot(id: string): Promise<void> {
    await this.api.focus('marker', 'hotspot-' + id, 800)
  }

  async focusCamera(id: string): Promise<void> {
    await this.api.focus('marker', 'camera-' + id, 400)
  }

  private key(p: number[]): string {
    return p[0] + ',' + p[1]
  }

  private at(p: number[]): number[] {
    return this.projected.get(this.key(p)) as number[]
  }

  /** 一次性把所有经纬度转成投影坐标 */
  private async projectAll(data: OverviewData): Promise<void> {
    const all: number[][] = []
    data.districts.forEach(d => {
      all.push(d.center)
      d.boundary.forEach(p => all.push(p))
    })
    data.hotspots.forEach(h => all.push(h.position))
    data.cameras.forEach(c => all.push(c.position))
    const todo = all.filter(p => !this.projected.has(this.key(p)))
    if (todo.length === 0) return
    const result = await this.api.project(todo)
    todo.forEach((p, i) => this.projected.set(this.key(p), result[i]))
  }

  private buildHeatmap(data: OverviewData) {
    const points = heatPointsAt(this.hour, data.districts, data.hotspots).map(p => {
      const c = this.at(p.position)
      return { id: p.id, coordinate: [c[0], c[1], c[2] ?? 0], radius: p.radius, heatValue: Math.min(HEAT_RANGE[1], p.value) }
    })
    const xs = points.map(p => p.coordinate[0])
    const ys = points.map(p => p.coordinate[1])
    const pad = 1200
    const [minZ, maxZ] = this.options.heatmapZ
    return {
      id: HEATMAP_ID,
      groupId: 'sc_heat',
      bbox: [Math.min(...xs) - pad, Math.min(...ys) - pad, minZ, Math.max(...xs) + pad, Math.max(...ys) + pad, maxZ],
      range: HEAT_RANGE,
      data: points
    }
  }
}
